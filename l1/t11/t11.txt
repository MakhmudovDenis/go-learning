	mutareSlice(slice)
	//fmt.Println(slice) //["vasya", "", "", "privet"], оригинальная строка до изменения mutareSlice
    fmt.Println(slice[2:]) //["vasya", "privet"]
}

func appendSlice(slice []string) []string {
	//slice = append(slice, "privet") //оригинальная строчка. Теперь возвращаем результат append
    return append(slice, "privet")
}
func mutareSlice(slice []string) {
	//slice[0] = "vasya" // оригинальная строчка
