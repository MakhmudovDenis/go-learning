package main

import (
	"fmt"
	"unsafe"
)

func main() {
	//1
	first := []int{1, 2, 3, 4, 5}
	first = nil
	fmt.Println("first:", first, ":", len(first), ":", cap(first))// nil слайс, len=0, cap=0

	//2
	second := []int{1, 2, 3, 4, 5}
	second = second[:0]
	fmt.Println("second:", second, ":", len(second), ":", cap(second)) non-nil, len=0, cap=5

	//3
	third := []int{1, 2, 3, 4, 5}
	clear(third)
	fmt.Println("third:", third, ":", len(third), ":", cap(third)) //[0,0,0,0,0], len=5, cap=5

	//4
	fourth := []int{1, 2, 3, 4, 5}
	clear(fourth[1:3])
	fmt.Println("fourth:", fourth, ":", len(fourth), ":", cap(fourth)) //[1,0,0,4,5], len=5, cap=5

	//5
	slice := make([]int, 3, 6) //[0,0,0], len=3, cap=6
	array := [3]int(slice[:3]) //{0,0,0}, len=3, cap=3
	slice[0] = 10 //[10,0,0], len=3, cap=6

	fmt.Println("slice = ", slice, len(slice), cap(slice)) // [10,0,0], len=3, cap==6
	fmt.Println("array =", array, len(array), cap(array)) //{0,0,0}, len=3, cap=3

	//6 В каких случаях Slice пустой или нулевой
	//1
	var data []string //nil
	fmt.Println("var data []string:")
	fmt.Printf("\tempty=%t nil=%t size=%d data=%p\n", len(data) == 0, data == nil, unsafe.Sizeof(data), unsafe.SliceData(data))
	//2
	data = []string(nil) //nil
	fmt.Println("data = []string(nil):")
	fmt.Printf("\tempty=%t nil=%t size=%d data=%p\n", len(data) == 0, data == nil, unsafe.Sizeof(data), unsafe.SliceData(data))
	//3
	data = []string{} //пустой
	fmt.Println("data = []string{}")
	fmt.Printf("\tempty=%t nil=%t size=%d data=%p\n", len(data) == 0, data == nil, unsafe.Sizeof(data), unsafe.SliceData(data))
	//4
	data = make([]string, 0) //пустой
	fmt.Println("data =make([]string,0)")
	fmt.Printf("\tempty=%t nil=%t size=%d data=%p\n", len(data) == 0, data == nil, unsafe.Sizeof(data), unsafe.SliceData(data))

	empty := struct{}{} //специальный адрес для всех пустых структур как zero-memory штука
	fmt.Println("empty struct address ", unsafe.Pointer(&empty))
}
